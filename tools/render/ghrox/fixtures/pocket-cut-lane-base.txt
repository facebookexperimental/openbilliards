export function makeGeom(id) {
  const spec = specFor(id);
  const k = spec.kernel, r = spec.render;
  // pocketless carom branch (feature 012): mirrors Table::from_carom_spec —
  // 4 linear cushions with endpoints AT the four corner points, zero
  // pockets; no jaw/pocket-derived quantities exist.
  if (k.pocketless) {
    return {
      id, kernel: k, render: r,
      L: k.length, W: k.width, CW: k.cushionWidth,
      pockets: [],
    };
  }
  // curved-pocket (CEBP template) branch — catalog v7 Amendment A1: mirrors
  // Table::from_curved_pocket_spec. Rails end at the jaw tangency points,
  // jaws are the sampled tangent two-arc composites (railPolysTemplate),
  // pockets are the template FALL circles. The pool populate() consumes the
  // fall discs through the same kernel-field names it uses for the 9-ft
  // entries (cornerPocketRadius / sidePocketDepth / sidePocketRadius + cd),
  // filled in here from the template construction.
  if (k.template) {
    const L = k.length, W = k.width, CW = k.cushionWidth;
    const corner = cornerCanonical(k.cornerScale);
    const middle = middleCanonical(k.middleScale);
    const cfo = CEBP.cornerFallOff * k.cornerScale;
    // catalog v16 (calibration B1): the cn8Molded corner family mirrors
    // the kernel's R82 construction — rails end at the departure Y_N,
    // capture disc = the Ø105 mouth circle about the drawing O_f
    // (±(L/2 + C), ±(W/2 + C) — the bed-hole/cammatch datum). `cd`
    // deliberately stays cfo (the slate-edge datum the dressed template
    // consumers key on); the pocket centres carry the drawing C.
    const cn8 = k.cornerFamily === 'cn8Molded';
    const cRf = cn8 ? CN8_KERNEL.fallR : corner.rf;
    const cOff = cn8 ? CN8_KERNEL.fallC : cfo;
    const kk = {
      ...k,
      cornerPocketRadius: cRf,
      sidePocketRadius: middle.rf,
      sidePocketDepth: CEBP.middleFallDepth * k.middleScale,
    };
    return {
      id, kernel: kk, render: r, L, W, CW,
      template: true, corner, middle,
      utc: cn8 ? CN8_KERNEL.dep : corner.ut, utm: middle.ut, cd: cfo,
      pockets: [
        { x: -L / 2 - cOff, z: -W / 2 - cOff, r: cRf, corner: true },
        { x: 0, z: -W / 2 - kk.sidePocketDepth, r: middle.rf, corner: false },
        { x: L / 2 + cOff, z: -W / 2 - cOff, r: cRf, corner: true },
        { x: -L / 2 - cOff, z: W / 2 + cOff, r: cRf, corner: true },
        { x: 0, z: W / 2 + kk.sidePocketDepth, r: middle.rf, corner: false },
        { x: L / 2 + cOff, z: W / 2 + cOff, r: cRf, corner: true },
      ],
    };
  }
  const L = k.length, W = k.width, CW = k.cushionWidth;
  const ca = (k.cornerFacingDeviationDeg + 45.0) * Math.PI / 180.0;
  const sa = k.sideFacingDeviationDeg * Math.PI / 180.0;
  const dc = k.cornerJawRadius / Math.tan((Math.PI / 2 + ca) / 2);
  const ds = k.sideJawRadius / Math.tan((Math.PI / 2 + sa) / 2);
  // true-mouth inversion (table.rs from_pocket_spec): recovers the pooltool
  // widths bit-exactly for the default spec
  const pwc = k.cornerMouth / Math.SQRT2 - dc;
  const sw = k.sideMouth - 2.0 * ds;
  // side-jaw back-endpoint rise: pooltool cw·sin(sa) quirk (default) vs the
  // exact facing line cw·tan(sa) (measured entries) — table.rs `sjb`
  const sjb = k.exactSideFacing ? CW * Math.tan(sa) : CW * Math.sin(sa);
  const cd = k.cornerPocketDepth / Math.SQRT2;
  return {
    id, kernel: k, render: r, L, W, CW, ca, sa, dc, ds, pwc, sw, sjb, cd,
    cosCa: Math.cos(ca), sinCa: Math.sin(ca), tanCa: Math.tan(ca),
    cosSa: Math.cos(sa), sinSa: Math.sin(sa),
    // pockets, kernel order lb lc lt rb rc rt (three.js frame)
    pockets: [
      { x: -L / 2 - cd, z: -W / 2 - cd, r: k.cornerPocketRadius, corner: true },
      { x: 0, z: -W / 2 - k.sidePocketDepth, r: k.sidePocketRadius, corner: false },
      { x: L / 2 + cd, z: -W / 2 - cd, r: k.cornerPocketRadius, corner: true },
      { x: -L / 2 - cd, z: W / 2 + cd, r: k.cornerPocketRadius, corner: true },
      { x: 0, z: W / 2 + k.sidePocketDepth, r: k.sidePocketRadius, corner: false },
      { x: L / 2 + cd, z: W / 2 + cd, r: k.cornerPocketRadius, corner: true },
    ],
  };
}

/** Inward polygon offset for a convex-ish closed outline (points [[x,z],...], metres): each vertex moves along
 *  the inward bisector by d. Used for the 3 mm apron skin ring (RB-05) and the trim strip's inner face. */
function roxOffsetPolygon(pts, d) {
  const n = pts.length; const out = [];
  let area = 0; for (let i = 0; i < n; i++) { const a = pts[i], b = pts[(i + 1) % n]; area += a[0] * b[1] - b[0] * a[1]; }
  const s = area > 0 ? 1 : -1;
  for (let i = 0; i < n; i++) {
    const p = pts[(i - 1 + n) % n], c = pts[i], q = pts[(i + 1) % n];
    const e1 = [c[0] - p[0], c[1] - p[1]], e2 = [q[0] - c[0], q[1] - c[1]];
    const l1 = Math.hypot(e1[0], e1[1]) || 1, l2 = Math.hypot(e2[0], e2[1]) || 1;
    const n1 = [-s * e1[1] / l1, s * e1[0] / l1], n2 = [-s * e2[1] / l2, s * e2[0] / l2];
    let bx = n1[0] + n2[0], bz = n1[1] + n2[1]; const bl = Math.hypot(bx, bz) || 1; bx /= bl; bz /= bl;
    const cosHalf = Math.max(Math.SQRT1_2 / 2, bx * n1[0] + bz * n1[1]); // mitre clamp at sharp vertices (cos 69°, a ratio, not a dimension)
    out.push([c[0] + bx * d / cosHalf, c[1] + bz * d / cosHalf]);
  }
  return out;
}

/** Rail-band section polygons (u across the rail from the nose, z above the bed; mm) for one rail family:
 *  the RB-04 dome shell carried to the 9-ft band (RC-03), the RC-01/RC-02 lip with its inner wall, and the
 *  RB-05 shelf ledge; split into the wood dome and the aluminium lip piece (RB-06) at the top-edge round (RB-04). */
function roxCapSections(fam) {
  const k = ROX_BAND_SCALE[fam], lip = ROX_LIP[fam], t = ROX_CAP_SHELL;
  const u9 = (u) => ROX_CUSHION_W + (u - ROX_CUSHION_W) * k; // RC-03 formula
  const domeTop = ROX_CAP_TOP.filter((p) => p[0] <= ROX_CAP_EDGE_ROUND_START).map(([u, z]) => [u9(u), z]);
  const roundTop = ROX_CAP_TOP.filter((p) => p[0] > ROX_CAP_EDGE_ROUND_START).map(([u, z]) => [u9(u), z]);
  const lipFace = u9(lip.face);
  const split = domeTop[domeTop.length - 1];

  // inner face runs RB-04 inner_face_z 38.528 → 41.037 (= 41.041 − 2.522 at the inner edge).
  const dome = [[inner, ROX_CAP_TOP_Z_INNER - t], [inner, ROX_CAP_TOP_Z_INNER], ...domeTop,
    ...domeTop.slice().reverse().map(([u, z]) => [u, z - t])];
  const wallU = lipFace - ROX_LIP_THICK; // RB-04 vertical_lip inner wall (thickness raw, A2-2)

/** Extrude a 2-D section (points [[u,z]] in mm) along a rail run, in scene metres. fam 'long': section plane
 *  z–y, run along x from a0 to a1 on the ±z rail (side); fam 'short': plane x–y, run along z on the ±x end.
 *  The basis is always right-handed (no mirrored windings). */
function roxExtrudeAlongRail(pts, fam, side, a0, a1, nose) {
  const shape = new THREE.Shape(pts.map(([u, z]) => new THREE.Vector2(u / 1000, z / 1000)));
  const len = a1 - a0;
  const geo = new THREE.ExtrudeGeometry(shape, { depth: len, bevelEnabled: false, curveSegments: 1 });
  const X = fam === 'long' ? new THREE.Vector3(0, 0, side) : new THREE.Vector3(side, 0, 0);
  const Y = new THREE.Vector3(0, 1, 0);
  const Z = new THREE.Vector3().crossVectors(X, Y);
  geo.applyMatrix4(new THREE.Matrix4().makeBasis(X, Y, Z));
  const along = fam === 'long' ? Z.x : Z.z;
  const runStart = along > 0 ? a0 : a1;
  if (fam === 'long') geo.translate(runStart, 0, side * nose); else geo.translate(side * nose, 0, runStart);
  geo.computeVertexNormals();
  return geo;
}

/** Plate transverse outline (across y, height z; mm, floor datum): RC-11 width by height, the arch = EN-03's
 *  polyline widened to RC-12, feet down to the foot-plate top. */
function roxPlateOutline(zBottom, zTop) {
  const halfW = (z) => {
    const T = ROX_PED_WIDTH;
    if (z <= T[0][0]) return T[0][1] / 2; // RC-11 foot line (z 100) carried down to the foot plate
    for (let i = 1; i < T.length; i++) if (z <= T[i][0]) { const t = (z - T[i - 1][0]) / (T[i][0] - T[i - 1][0]); return (T[i - 1][1] + (T[i][1] - T[i - 1][1]) * t) / 2; }
    const n = T.length; const slope = (T[n - 1][1] - T[n - 2][1]) / (T[n - 1][0] - T[n - 2][0]); // RC-11: the 500 → 550 slope extrapolated to the top
    return (T[n - 1][1] + slope * (z - T[n - 1][0])) / 2;
  };
  const zScale = ROX_PED_ARCH_APEX / ROX_PED_ARCH_MODEL_APEX;
  const archHalf = (z) => {
    const zm = z / zScale; const P = ROX_PED_ARCH_MODEL;
    if (zm <= P[0][0]) return P[0][1] * ROX_PED_ARCH_SPAN_SCALE;
    for (let i = 1; i < P.length; i++) if (zm <= P[i][0]) { const t = (zm - P[i - 1][0]) / (P[i][0] - P[i - 1][0]); return (P[i - 1][1] + (P[i][1] - P[i - 1][1]) * t) * ROX_PED_ARCH_SPAN_SCALE; }
    return 0;
  };
  const pts = [];
  const zs = []; for (let z = zBottom; z < zTop; z += 12) zs.push(z); zs.push(zTop); // 12 mm sampling of the RC-11 taper
  for (const z of zs) pts.push([-halfW(z), z]);
  for (let i = zs.length - 1; i >= 0; i--) pts.push([halfW(zs[i]), zs[i]]);
  pts.push([archHalf(zBottom), zBottom]);
  const zA = []; for (let z = zBottom; z < ROX_PED_ARCH_APEX; z += 8) zA.push(z); zA.push(ROX_PED_ARCH_APEX);
  for (const z of zA) pts.push([archHalf(z), z]);
  for (let i = zA.length - 2; i >= 0; i--) pts.push([-archHalf(zA[i]), zA[i]]);
  return { pts, halfW, archHalf };
}

/** One transverse pedestal plate + two foot plates (+ the beam once, on side = +1). side = ±1 = the ±x end.
 *  The RC-11/RC-12 transverse outline is extruded 1 unit and bent onto the RC-13 bow circles. */
function roxPedestal(group, dm, side, mats) {
  const zFoot = ROX_FOOT.height, zTop = ROX_BODY_UNDER;
  const outline = roxPlateOutline(zFoot, zTop);

function roxLerp(tab, x) {
  if (x <= tab[0][0]) return tab[0][1];
  for (let i = 1; i < tab.length; i++) if (x <= tab[i][0]) { const t = (x - tab[i - 1][0]) / (tab[i][0] - tab[i - 1][0]); return tab[i - 1][1] + (tab[i][1] - tab[i - 1][1]) * t; }
  return tab[tab.length - 1][1];
}

/** Non-indexed grid surface from rows of equal length ([[Vector3,...], ...]); flat normals. */
function roxGrid(rows, orient = null) {
  const pos = [], idx = []; const m = rows[0].length;
  for (const row of rows) for (const v of row) pos.push(v.x, v.y, v.z);
  for (let i = 1; i < rows.length; i++) for (let j = 1; j < m; j++) {
    const a = (i - 1) * m + (j - 1), b = (i - 1) * m + j, c = i * m + j, d = i * m + (j - 1);
    idx.push(a, b, c, a, c, d);
  }
  if (orient) { // 'up': the visible face is the top (+y); 'out': the visible face looks away from the table centre
    let score = 0;
    for (let k = 0; k < idx.length; k += 3) {
      const A = new THREE.Vector3(pos[3 * idx[k]], pos[3 * idx[k] + 1], pos[3 * idx[k] + 2]), B = new THREE.Vector3(pos[3 * idx[k + 1]], pos[3 * idx[k + 1] + 1], pos[3 * idx[k + 1] + 2]), C = new THREE.Vector3(pos[3 * idx[k + 2]], pos[3 * idx[k + 2] + 1], pos[3 * idx[k + 2] + 2]);
      const n = new THREE.Vector3().subVectors(B, A).cross(new THREE.Vector3().subVectors(C, A)); const c = new THREE.Vector3().addVectors(A, B).add(C);
      score += orient === 'up' ? n.y : (n.x * c.x + n.z * c.z);
    }
    if (score < 0) for (let k = 0; k < idx.length; k += 3) { const t = idx[k + 1]; idx[k + 1] = idx[k + 2]; idx[k + 2] = t; }
  }
  const g = new THREE.BufferGeometry(); g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3)); g.setIndex(idx); g.computeVertexNormals(); return g;
}
/** Smallest positive ray parameter t (mm) where P + t·dir crosses the closed polyline pts ([[x,z],...], mm); Infinity if none. */
function roxRayPolyline(P, dir, pts) {
  let best = Infinity, seg = -1;
  for (let i = 1; i < pts.length; i++) {
    const [x1, z1] = pts[i - 1], [x2, z2] = pts[i]; const ex = x2 - x1, ez = z2 - z1;
    const den = dir[0] * ez - dir[1] * ex; if (Math.abs(den) < 1e-9) continue; // roxlit: construction
    const t = ((x1 - P[0]) * ez - (z1 - P[1]) * ex) / den; const w = ((x1 - P[0]) * dir[1] - (z1 - P[1]) * dir[0]) / den;
    if (t > 0 && w >= 0 && w <= 1 && t < best) { best = t; seg = i - 1; }
  }
  return [best, seg];
}

/** Smallest positive t where P + t·u crosses the segment A→B (2-D, mm); Infinity if none. */
function roxRaySeg(P, u, A, B) { // roxlit: construction
  const ex = B[0] - A[0], ey = B[1] - A[1]; const den = u[0] * ey - u[1] * ex; if (Math.abs(den) < 1e-12) return Infinity; // roxlit: construction
  const t = ((A[0] - P[0]) * ey - (A[1] - P[1]) * ex) / den; const w = ((A[0] - P[0]) * u[1] - (A[1] - P[1]) * u[0]) / den;
  return (t > 0 && w >= -1e-9 && w <= 1 + 1e-9) ? t : Infinity; // roxlit: construction
}
/** Intersection of the line A→B with the line C→D (2-D); null when parallel. */
function roxLineLine(A, B, C, D) { // roxlit: construction
  const ex = B[0] - A[0], ey = B[1] - A[1], fx = D[0] - C[0], fy = D[1] - C[1]; const den = ex * fy - ey * fx; if (Math.abs(den) < 1e-15) return null; // roxlit: construction
  const t = ((C[0] - A[0]) * fy - (C[1] - A[1]) * fx) / den; return [A[0] + t * ex, A[1] + t * ey];
}

 *  clamped to CU-11 (cushion_body_top 41.11) exactly as the shared clamp did. Names / order stay `dm-cushion-<i>`. */
function roxCushionGeom(G, poly, prof) {
  const M = poly.length; const A = poly[0], B = poly[M - 1]; // back corners (the last edge B→A is the fixed slate-edge face)
  let iN0 = 1, best = -1; // the nose run = the longest front edge
  for (let i = 1; i < M - 2; i++) { const l = Math.hypot(poly[i + 1][0] - poly[i][0], poly[i + 1][1] - poly[i][1]); if (l > best) { best = l; iN0 = i; } }
  const iN1 = iN0 + 1; const N0 = poly[iN0], N1 = poly[iN1];
  let area2 = 0; for (let i = 0; i < M; i++) { const [x1, y1] = poly[i], [x2, y2] = poly[(i + 1) % M]; area2 += x1 * y2 - x2 * y1; }
  const sgn = area2 > 0 ? 1 : -1;
  const nl = Math.hypot(N1[0] - N0[0], N1[1] - N0[1]); const nd = [(N1[0] - N0[0]) / nl, (N1[1] - N0[1]) / nl]; const nn = [-nd[1] * sgn, nd[0] * sgn]; // inward normal of the nose edge
  const rows = prof.rows; const capZ = ROX_CUSHION_TOP / 1000; // CU-11
  const Q0 = roxLineLine(A, poly[1], N0, N1), Q1 = roxLineLine(B, poly[M - 2], N0, N1); // the sharp corners nose line ∩ facing plane (the kernel's jaw round is tangent to both)
  const ringAt = (o) => { // every column keeps its index so the grid is rectangular; the back corners never move
    if (o <= 0) return poly.map((q) => [q[0], q[1]]);
    const P0 = [N0[0] + o * nn[0], N0[1] + o * nn[1]], P1 = [N1[0] + o * nn[0], N1[1] + o * nn[1]]; // the inset nose line
    const R0 = roxLineLine(A, poly[1], P0, P1) ?? P0, R1 = roxLineLine(B, poly[M - 2], P0, P1) ?? P1; // fixed facing planes ∩ inset nose line
    const d0 = Q0 && R0 ? [R0[0] - Q0[0], R0[1] - Q0[1]] : [0, 0], d1 = Q1 && R1 ? [R1[0] - Q1[0], R1[1] - Q1[1]] : [0, 0]; // the corner slides ALONG the facing plane
    const out = []; // the jaw round (tangent to the nose line and to the facing) rides with its corner: radius and tangency are unchanged, only the corner moves
    for (let j = 0; j < M; j++) out.push(j === 0 ? [A[0], A[1]] : j === M - 1 ? [B[0], B[1]] : j <= iN0 ? [poly[j][0] + d0[0], poly[j][1] + d0[1]] : [poly[j][0] + d1[0], poly[j][1] + d1[1]]);
    return out;
  };
  const rings = rows.map((r) => ringAt(r.o)); const zs = rows.map((r) => Math.min(r.z, capZ));
  const uBase = [0]; for (let i = 0; i < M; i++) { const [x1, y1] = poly[i], [x2, y2] = poly[(i + 1) % M]; uBase.push(uBase[i] + Math.hypot(x2 - x1, y2 - y1)); }
  const vProf = [0]; for (let k = 1; k < rows.length; k++) vProf.push(vProf[k - 1] + Math.hypot(rows[k].o - rows[k - 1].o, rows[k].z - rows[k - 1].z));
  const pos = [], uv = [], idx = [];
  for (let k = 0; k < rows.length; k++) for (let j = 0; j < M; j++) { const [kx, ky] = rings[k][j]; pos.push(ky - G.L / 2, zs[k], kx - G.W / 2); uv.push(uBase[j], vProf[k]); }
  const same = (k, j, k2, j2) => rings[k][j][0] === rings[k2][j2][0] && rings[k][j][1] === rings[k2][j2][1] && zs[k] === zs[k2];
  for (let k = 0; k + 1 < rows.length; k++) for (let j = 0; j < M; j++) { // M edges per ring: the last one (B → A) is the fixed slate-edge back face, lofted like the rest
    const j1 = (j + 1) % M; const a = k * M + j, b = k * M + j1, c = (k + 1) * M + j, d = (k + 1) * M + j1;
    const t1 = !(same(k, j, k, j1) || same(k, j1, k + 1, j) || same(k, j, k + 1, j)), t2 = !(same(k, j1, k + 1, j1) || same(k + 1, j1, k + 1, j) || same(k, j1, k + 1, j));
    if (sgn > 0) { if (t1) idx.push(a, b, c); if (t2) idx.push(b, d, c); } else { if (t1) idx.push(a, c, b); if (t2) idx.push(b, c, d); }
  }
  const top = rings[rings.length - 1]; const cap = top.filter((q, j) => j === 0 || Math.hypot(q[0] - top[j - 1][0], q[1] - top[j - 1][1]) > 1e-9); // roxlit: construction — the top ring, duplicates dropped; its ends lie on the two vertical planes
  const contour = cap.map(([kx, ky]) => new THREE.Vector2(ky - G.L / 2, -(kx - G.W / 2)));
  const base = pos.length / 3; for (const v of contour) { pos.push(v.x, zs[zs.length - 1], -v.y); uv.push(v.x, v.y); }
  for (const t of THREE.ShapeUtils.triangulateShape(contour, [])) idx.push(base + t[0], base + t[1], base + t[2]);
  const g = new THREE.BufferGeometry(); g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3)); g.setAttribute('uv', new THREE.Float32BufferAttribute(uv, 2)); g.setIndex(idx); g.computeVertexNormals(); return g;
}
/** Plan footprint of one cushion END in the pocket's own frame: the strip between its nose line and its back line, closed by the
 *  facing segment J→B (vertical after roxCushionGeom) and the jaw round N→J. `f(x)` maps a frame point to [along, behind] with
 *  `along` positive INTO the rail (away from the pocket) and `behind` positive behind that rail's nose line. All from G (the
 *  kernel mirror railPolysKernel already uses: mouth, jaw radius, facing angle, cushion width). */
function roxCushionEnd(G, kind) {
  const CW = G.CW * 1000;
  if (kind === 'side') { const h = G.sw / 2 * 1000, ds = G.ds * 1000; return { N: [h + ds, 0], J: [h - ds * G.sinSa, ds * G.cosSa], B: [h - G.sjb * 1000, CW], CW }; }
  const pwc = G.pwc * 1000, dc = G.dc * 1000; return { N: [pwc + dc, 0], J: [pwc - dc * G.sinCa, dc * G.cosCa], B: [pwc - CW * G.tanCa, CW], CW };
}
const roxInsideEnd = (E, along, behind) => { // roxlit: construction
  if (behind < 0 || behind > E.CW) return false;
  const [Pa, Pb] = behind <= E.J[1] ? [E.N, E.J] : [E.J, E.B]; const t = (behind - Pa[1]) / ((Pb[1] - Pa[1]) || 1e-9); // roxlit: construction
  return along >= Pa[0] + t * (Pb[0] - Pa[0]) - 1e-9; // roxlit: construction
};
/** The MODEL's cushion end in the same frame: our facing direction (CP-02 / SP-02 are the model's own planar cheek fits) through the
 *  model's back-top corner (ROX_*_MODEL_FACE). Points inside it and outside ours are the mouth-ruling residual strip the derived top
 *  closes (the model's rubber is 2.7-4.6 mm longer per side than the MFR mouth allows; RC-29). */
const roxInsideModelEnd = (E, face, along, behind) => { // roxlit: construction
  if (behind < 0 || behind > E.CW) return false;
  const fx = E.B[0] - E.J[0], fy = E.B[1] - E.J[1]; return along >= face[0] + (behind - face[1]) * fx / fy - 1e-9; // roxlit: construction
};
/** Covered runs along a ray (mm): the derived furniture top exists where the ray is (i) outside every cushion footprint and
 *  (ii) over the bed hole or behind a back line — never over the visible cloth shelf. Sampled every half millimetre. */
function roxRayRuns(t0, t1, classify) { // classify(t) -> 'top' | 'cushion' | 'cloth' ; roxlit: construction
  const runs = []; let cur = null; const step = 1 / 2;
  for (let t = t0; t <= t1 + 1e-9; t += step) { const c = classify(t); if (c === 'top') { if (!cur) cur = [t, t]; else cur[1] = t; } else if (cur) { runs.push(cur); cur = null; if (c === 'cloth') break; } } // roxlit: construction
  if (cur) runs.push(cur); return runs;
}
function roxPocketCut(group, dm, G, p, kind, mats, clothMaterial) {
  const Rh = p.r * 1000 + ROX_BED_HOLE_REVEAL; const Rf = Rh - ROX_CUT_TOP_R;
  const prof = [[Rh, 0]];
  for (let k = 1; k <= 6; k++) { const t = Math.PI / 2 * k / 6; prof.push([Rh - ROX_CUT_TOP_R * Math.sin(t), -ROX_CUT_TOP_R * (1 - Math.cos(t))]); } // roxlit: construction
  prof.push([Rf, ROX_CUT_BOTTOM_Z]);
  for (let k = 1; k <= 6; k++) { const t = Math.PI / 2 * k / 6; prof.push([Rh - ROX_CUT_BOTTOM_R * Math.cos(t), ROX_CUT_BOTTOM_Z - ROX_CUT_BOTTOM_R * Math.sin(t)]); } // roxlit: construction
  // A3: the cloth roll-over is a ring of angular segments about the shared hole (RC-22); the segments whose top edge lies under a
  // cushion end are dropped — the cushion's vertical end face now stands on the hole's own footprint there (the model's notch runs
  // under its rubber with no roll-over), and a lathe through the rubber read as a 50-60 mm crossing in the C14 bed slice
  const L = G.L * 1000, W = G.W * 1000; const E = roxCushionEnd(G, p.corner ? 'corner' : 'side'); const cx = p.x * 1000, cz = p.z * 1000;
  const inEnd = (x, z) => { if (!p.corner) return roxInsideEnd(E, Math.abs(x), (z - Math.sign(p.z) * W / 2) * Math.sign(p.z));
    const a = (x - Math.sign(p.x) * L / 2) * Math.sign(p.x), b = (z - Math.sign(p.z) * W / 2) * Math.sign(p.z); return roxInsideEnd(E, -a, b) || roxInsideEnd(E, -b, a); };
  // the seen cup (RC-32 centres, CP-12 / SP-12 radii): the model's notch is cut around its cup wall, so the roll-over has no segment
  // where the cup wall stands — the wall itself is the edge there (the two circles are 0.1-1.2 mm apart at the corner's outward diagonal)
  const Sm = p.corner ? [Math.sign(p.x) * (L / 2 + ROX_CORNER_CUP_U / Math.SQRT2), Math.sign(p.z) * (W / 2 + ROX_CORNER_CUP_U / Math.SQRT2)] : [0, Math.sign(p.z) * (W / 2 + ROX_SIDE_CUP_CENTRE)];
  const Sr = (p.corner ? ROX_CUP_R_IN.corner : ROX_CUP_R_IN.side) + ROX_CUP_WALL + ROX_A3_EPS;
  const inCup = (x, z) => Math.hypot(x - Sm[0], z - Sm[1]) < Sr;
  const rows = []; const nA = 96; // roxlit: construction
  for (let k = 0; k <= nA; k++) { const t = 2 * Math.PI * k / nA; const ux = Math.cos(t), uz = Math.sin(t); // roxlit: construction
    const px = cx + Rh * ux, pz = cz + Rh * uz, qx = cx + Rf * ux, qz = cz + Rf * uz;
    rows.push(inEnd(px, pz) || inCup(px, pz) || inCup(qx, qz) ? null : prof.map(([r, y]) => new THREE.Vector3((cx + r * ux) / 1000, y / 1000, (cz + r * uz) / 1000))); }
  if (rows[nA] && rows[0]) rows[nA] = rows[0];
  const runs = []; let cur = []; for (const r of rows) { if (r) cur.push(r); else if (cur.length) { runs.push(cur); cur = []; } } if (cur.length) runs.push(cur);
  if (runs.length > 1 && rows[0] && rows[nA]) { runs[0] = runs[runs.length - 1].concat(runs[0]); runs.pop(); } // the ring closes through angle 0
  runs.forEach((run, k) => { if (run.length > 1) group.add(dm(new THREE.Mesh(roxGrid(run), clothMaterial), k === 0 ? `pocket-rox-${kind}-cut` : `pocket-rox-${kind}-cut-${k}`)); });
  const pan = new THREE.Mesh(new THREE.CircleGeometry(Rh / 1000, 48), mats.well); pan.rotation.x = -Math.PI / 2; pan.position.set(p.x, ROX_PAN_Z[kind.startsWith('side') ? 'side' : 'corner'] / 1000, p.z); // roxlit: construction
  group.add(dm(pan, `pocket-rox-${kind}-pan`));
}

/** One corner pocket assembly: cup C-liner + closed bucket, flange sheet, aluminium casting (dome mitre + lip corner).
 *  p = kernel pocket (scene metres). A3-PLAN: the SEEN liner hangs on the MODEL's own cup centre (CP-12 a3, ROX_CORNER_CUP_U along
 *  the diagonal from the nose-line intersection); the kernel axis p keeps physics + the bed hole. The C ends where its outer circle
 *  meets the cushion END FACE of the same height (vertical after roxCushionGeom); with the MFR corner mouth kept (CP-01, 114.3)
 *  that intersection does not exist, so the C ends on the cushion BACK LINE — the line the model's own ends sit on — and the gap
 *  to the kernel back corner is the mouth-ruling residual the flange closes (RC-29). Every top sheet (flange, ears, between-ends)
 *  is a ray from the cup centre clipped against the real cushion footprint and the bed hole (roxRayRuns) — no band constant. */
function roxCornerPocket(group, dm, G, p, mats, tag) {
  const L = G.L * 1000, W = G.W * 1000, sx = Math.sign(p.x), sz = Math.sign(p.z);
  const Ba = (G.CW + G.render.frameWidthEnd) * 1000, Bb = (G.CW + G.render.frameWidth) * 1000; // RB-02 / RB-01 bands: the lip lines
  const aOf = (x) => (x - sx * L / 2) * sx, bOf = (z) => (z - sz * W / 2) * sz; // beyond the short / long nose lines, mm
  const xOf = (a) => sx * (L / 2 + a), zOf = (b) => sz * (W / 2 + b);
  const d = [sx / Math.SQRT2, sz / Math.SQRT2], n = [-sx / Math.SQRT2, sz / Math.SQRT2]; // outward diagonal, and toward the long rail
  const Pm = [xOf(ROX_CORNER_CUP_U / Math.SQRT2), zOf(ROX_CORNER_CUP_U / Math.SQRT2)]; // RC-32: the model's cup centre on the pocket diagonal
  const Rin = ROX_CUP_R_IN.corner, Rout = Rin + ROX_CUP_WALL; // CP-12 liner radius
  const dir = (deg) => { const r = deg * Math.PI / 180; return [Math.cos(r) * d[0] + Math.sin(r) * n[0], Math.cos(r) * d[1] + Math.sin(r) * n[1]]; };
  const V = (r, deg, z) => { const u = dir(deg); return new THREE.Vector3((Pm[0] + r * u[0]) / 1000, z / 1000, (Pm[1] + r * u[1]) / 1000); };
  const rim = (deg) => roxLerp(ROX_CORNER_RIM, Math.abs(deg));
  // cushion ends in this pocket's frame (kernel mirror, the same numbers railPolysKernel uses) and the shared bed hole (RC-22)
  const E = roxCushionEnd(G, 'corner');
  const inCushion = (x, z) => roxInsideEnd(E, -aOf(x), bOf(z)) || roxInsideEnd(E, -bOf(z), aOf(x)); // long-rail end, short-rail end
  const hole = { x: p.x * 1000, z: p.z * 1000, R: p.r * 1000 + ROX_BED_HOLE_REVEAL };
  const inHole = (x, z) => Math.hypot(x - hole.x, z - hole.z) < hole.R;
  const inModel = (x, z) => roxInsideModelEnd(E, ROX_CORNER_MODEL_FACE, -aOf(x), bOf(z)) || roxInsideModelEnd(E, ROX_CORNER_MODEL_FACE, -bOf(z), aOf(x));
  // top where: behind a back line (rail region), inside the model's rubber but outside ours (the residual strip), or over the bed hole
  // behind a nose line; cloth (the shelf) in front of the nose lines or beside the cup where the model itself shows cloth
  // no cut void in front of the liner: the model's slab cut (CP-14m, an arc R 52.6 about the cup centre spanning 125 deg about the front
  // diagonal) lies UNDER its cloth — the model's bed-top sheet spans the cut to the cup wall (its own triangulation fan), and on the real
  // table the cloth wraps the slab edge down into the pocket — so from above the top over the cut is cloth (the shelf), never the pan.
  // (An earlier 'in front of both nose lines' void split the mouth into cloth slivers around the nose-line crossing 45.05 from the cup
  // centre and dropped a ray's real shelf when its first run was such a sliver; a per-ray 125 deg void showed the pan where the model
  // shows cloth: plan-law row (b) 44.8 / 45.0 vs the model's 17.0 / 21.2.)
  const inCup = (x, z) => Math.hypot(x - Pm[0], z - Pm[1]) < Rout; // the cup's own opening — never a top (the model's corner opening is open all round its C)
  // kinds: our rubber; the cup opening ('void'); 'strip' = the model's rubber or the rail region (flange sheet); 'shelf' = over the shared
  // bed hole where the model's slate shelf stands (cloth patch at bed level); 'void' also over the hole inside the model's own cut (left
  // open, as the model); 'cloth' = the shared bed sheet is the right top and the sheet stops
  // the model-rubber-minus-ours wedge runs along the whole jaw face; only its part behind the model's back-top corner (CP-12 C end,
  // 47.26 behind the nose line) is the between-ends top (flange); further forward the wedge is shelf beside a shorter rubber — cloth
  const inStrip = (x, z) => inModel(x, z) && (bOf(z) >= ROX_CORNER_MODEL_FACE[1] || aOf(x) >= ROX_CORNER_MODEL_FACE[1]);
  const kindAt = (x, z) => inCushion(x, z) ? 'cushion' : inCup(x, z) ? 'void' : (aOf(x) >= E.CW || bOf(z) >= E.CW || inStrip(x, z)) ? 'strip' : inHole(x, z) ? 'shelf' : 'cloth';
  const classify = (x, z) => { const k = kindAt(x, z); return k === 'strip' || k === 'shelf' ? 'top' : k; };
  // (0) where the C ends: its outer circle ∩ the long-rail facing segment (J→B, symmetric for the short rail) if that exists, else the back line
  const Cf = [-aOf(Pm[0]), bOf(Pm[1])]; // cup centre in the long-rail end's frame (along, behind)
  let phiEnd = null; {
    const ex = E.B[0] - E.J[0], ey = E.B[1] - E.J[1]; const fx = E.J[0] - Cf[0], fy = E.J[1] - Cf[1];
    const qa = ex * ex + ey * ey, qb = 2 * (fx * ex + fy * ey), qc = fx * fx + fy * fy - Rout * Rout; const disc = qb * qb - 4 * qa * qc; // roxlit: construction
    if (disc >= 0) for (const sg of [-1, 1]) { const w = (-qb + sg * Math.sqrt(disc)) / (2 * qa); if (w >= 0 && w <= 1) { const al = E.J[0] + w * ex, be = E.J[1] + w * ey; phiEnd = Math.atan2((be - Cf[1]) - (-(al) - (-Cf[0])), (be - Cf[1]) + (-(al) - (-Cf[0]))) * 180 / Math.PI; } }
  }
  const contact = phiEnd !== null;
  // no contact under the MFR mouth (RC-29): the C keeps the model's OWN span (CP-12 ±114.9, its ends on its cushion back-top corners);
  // the outer circle ∩ back line would be ±114.0 — the two agree to 0.7 mm, the model's is the read
  const span = contact ? Math.abs(phiEnd) : ROX_CUP_SPAN; // the C is symmetric about the diagonal: ±span from the outward diagonal
  // (1) cup: open C from the floor to the rim, closed ring below the transition, floor annulus with the drain
  const rows = [];
  const nC = Math.ceil(2 * span / 3); // ~3° steps landing exactly on both ends // roxlit: construction
  for (let k = 0; k <= nC; k++) { const deg = -span + 2 * span * k / nC; rows.push([V(Rin, deg, ROX_CUP_OPEN_Z), V(Rin, deg, rim(deg)), V(Rout, deg, rim(deg)), V(Rout, deg, ROX_CUP_OPEN_Z)]); }
  // closed bucket below the transition (PK-03): the back arc continues the C, the front arc is set toward the table; the two
  // meet where the circles intersect (solved here, no authored angle)
  const B = ROX_CORNER_BUCKET; const dd = B.backOff - B.frontOff; const cosJ = (dd * dd + B.frontR ** 2 - B.backR ** 2) / (2 * dd * B.frontR);
  const thJ = Math.acos(Math.max(-1, Math.min(1, cosJ))) * 180 / Math.PI; // front-arc angle (from the outward diagonal) at the junction
  const jx = B.frontOff + B.frontR * Math.cos(thJ * Math.PI / 180), jy = B.frontR * Math.sin(thJ * Math.PI / 180);
  const phiJ = Math.atan2(jy, jx - B.backOff) * 180 / Math.PI; // back-arc angle at the same point
  const arcPt = (off, deg, r, z) => { const t = deg * Math.PI / 180; return new THREE.Vector3((Pm[0] + off * d[0] + r * (Math.cos(t) * d[0] + Math.sin(t) * n[0])) / 1000, z / 1000, (Pm[1] + off * d[1] + r * (Math.cos(t) * d[1] + Math.sin(t) * n[1])) / 1000); };
  const rowsB = [];
  const zB = ROX_CUP_OPEN_Z - 1 / 10; // the bucket's top ring sits a tenth of a millimetre under the C's bottom edge (coincident rings read as crossings)
  const nB1 = Math.ceil(2 * phiJ / 3), nB2 = Math.ceil((360 - 2 * thJ) / 3); // roxlit: construction
  for (let k = 0; k <= nB1; k++) { const deg = -phiJ + 2 * phiJ * k / nB1; rowsB.push([arcPt(B.backOff, deg, B.backR, ROX_CUP_FLOOR_Z), arcPt(B.backOff, deg, B.backR, zB), arcPt(B.backOff, deg, B.backR + ROX_CUP_WALL, zB), arcPt(B.backOff, deg, B.backR + ROX_CUP_WALL, ROX_CUP_FLOOR_Z)]); }
  for (let k = 0; k <= nB2; k++) { const deg = thJ + (360 - 2 * thJ) * k / nB2; rowsB.push([arcPt(B.frontOff, deg, B.frontR, ROX_CUP_FLOOR_Z), arcPt(B.frontOff, deg, B.frontR, zB), arcPt(B.frontOff, deg, B.frontR + ROX_CUP_WALL, zB), arcPt(B.frontOff, deg, B.frontR + ROX_CUP_WALL, ROX_CUP_FLOOR_Z)]); }
  const rowsF = [];
  for (let deg = 0; deg <= 360 + 1e-6; deg += 6) { const back = deg <= phiJ || deg >= 360 - phiJ; const fd = deg <= 180 ? Math.max(deg, thJ) : Math.min(deg, 360 - thJ); rowsF.push([V(ROX_CUP_DRAIN_R.corner, deg, ROX_CUP_FLOOR_Z), back ? arcPt(B.backOff, deg, B.backR, ROX_CUP_FLOOR_Z) : arcPt(B.frontOff, fd, B.frontR, ROX_CUP_FLOOR_Z)]); } // roxlit: construction
  const rowsT = []; // transition seat at the open_at height (CP-12 open_C_to_closed_bucket_transition): the C's open lower edge sits on the bucket, so the ledge runs from the bucket's outer wall out to the C circle all round — the PK-03 compound arcs run inside the C away from the diagonal
  const seatRow = (q) => { const rq = Math.hypot(q.x * 1000 - Pm[0], q.z * 1000 - Pm[1]); // roxlit: construction
    const ang = Math.atan2((q.x * 1000 - Pm[0]) * n[0] + (q.z * 1000 - Pm[1]) * n[1], (q.x * 1000 - Pm[0]) * d[0] + (q.z * 1000 - Pm[1]) * d[1]) * 180 / Math.PI;
    const outerR = Math.abs(ang) <= span ? Rin : Rout; // within the C: up to its inner face; past the C ends (no wall there): up to the outer circle
    return rq < outerR - 1 / 10 ? [q, V(outerR, ang, zB)] : null; }; // roxlit: construction
  for (let k = 0; k <= nB1; k++) { const deg = -phiJ + 2 * phiJ * k / nB1; rowsT.push(seatRow(arcPt(B.backOff, deg, B.backR + ROX_CUP_WALL, zB))); }
  for (let k = 0; k <= nB2; k++) { const deg = thJ + (360 - 2 * thJ) * k / nB2; rowsT.push(seatRow(arcPt(B.frontOff, deg, B.frontR + ROX_CUP_WALL, zB))); }
  rowsT.push(rowsT[0]);
  { const runsT = []; let cur = []; for (const r of rowsT) { if (r) cur.push(r); else if (cur.length) { runsT.push(cur); cur = []; } } if (cur.length) runsT.push(cur);
    runsT.forEach((run, k) => { if (run.length > 1) group.add(dm(new THREE.Mesh(roxGrid(run, 'up'), mats.well), `pocket-rox-corner-${tag}-cup-seat-${k}`)); }); }
  group.add(dm(new THREE.Mesh(roxGrid(rows), mats.well), `pocket-rox-corner-${tag}-cup-c`));
  group.add(dm(new THREE.Mesh(roxGrid([[rows[0][0], rows[0][3]], [rows[0][1], rows[0][2]]]), mats.well), `pocket-rox-corner-${tag}-cup-c-end-0`)); // the C's end faces (CP-12 C_end_points)
  group.add(dm(new THREE.Mesh(roxGrid([[rows[rows.length - 1][0], rows[rows.length - 1][3]], [rows[rows.length - 1][1], rows[rows.length - 1][2]]]), mats.well), `pocket-rox-corner-${tag}-cup-c-end-1`));
  group.add(dm(new THREE.Mesh(roxGrid(rowsB), mats.well), `pocket-rox-corner-${tag}-cup-bucket`));
  group.add(dm(new THREE.Mesh(roxGrid(rowsF), mats.well), `pocket-rox-corner-${tag}-cup-floor`));
  // ray helper: where a ray from the cup centre leaves the corner square (a ≥ CU-04 or b ≥ CU-04) — the casting's inner limit
  const exitT = (u) => { const da = u[0] * sx, db = u[1] * sz; const ta = da > 1e-9 ? (ROX_CUSHION_BACK - aOf(Pm[0])) / da : Infinity, tb = db > 1e-9 ? (ROX_CUSHION_BACK - bOf(Pm[1])) / db : Infinity; return Math.max(0, Math.min(ta, tb)); }; // roxlit: construction
  const hs = (deg) => (deg < ROX_CORNER_HS[0][0] || deg > ROX_CORNER_HS[ROX_CORNER_HS.length - 1][0]) ? null : roxLerp(ROX_CORNER_HS, deg);
  // (2) flange: from the C's outer face out to the horseshoe, clipped by the cushion footprints and the bed hole; a ray that passes
  // THROUGH a cushion end resumes behind it (the ear: CP-15 inner end face 41.4 on both sides of each band); a ray past the C end
  // belongs to the between-ends strip only while it still meets a cushion end face — the open throat gets no sheet
  const zFl = (deg, R) => { // CP-15c profile re-based from the model rim radius onto the liner one, scaled by this ray's rim height
    const Rm = R - (Rin - ROX_CORNER_FLANGE[0][0]); const zp = roxLerp(ROX_CORNER_FLANGE, Rm);
    const g = Math.max(0, (zp - ROX_FLANGE_FLAT_Z) / (ROX_CORNER_FLANGE[0][1] - ROX_FLANGE_FLAT_Z));
    return ROX_FLANGE_FLAT_Z + (rim(deg) - ROX_FLANGE_FLAT_Z) * g + Math.min(0, zp - ROX_FLANGE_FLAT_Z);
  };
  const sheetRows = [[], [], []]; // [flange rows, ear rows, shelf rows] per ray angle (null where absent)
  const fanAngles = []; // half-degree rays round the full circle, tenth-degree rays within four degrees of the two C ends (where the sheets meet the cushion back line); the classifier decides where a sheet exists
  for (let k = 0; k <= 2 * 360; k++) fanAngles.push(360 * k / (2 * 360) - 180); // roxlit: construction
  for (const e of [-span, span]) for (let dd = 0 - 4; dd <= 4 + 1e-9; dd += 1 / 10) fanAngles.push(e + dd); // roxlit: construction
  fanAngles.sort((p, q) => p - q);
  for (const deg of fanAngles) { const u = dir(deg); const h = hs(deg);
    const Ro = h ?? Rout; const beyond = Math.abs(deg) > span; // past the C end
    if (Ro <= Rout + 1e-6 && !beyond) { sheetRows[0].push(null); sheetRows[1].push(null); sheetRows[2].push(null); continue; } // roxlit: construction
    const far = h ?? 400; // roxlit: construction (past the horseshoe table the ray can only serve the between-ends strip, which ends on the cushion)
    const R0 = beyond ? ROX_CUP_DRAIN_R.corner : Rout; // past the C end the ray starts at the drain: the model's slate shelf runs over the cup's own front, and there is no wall in the way
    const runs = roxRayRuns(R0, far, (t) => classify(Pm[0] + t * u[0], Pm[1] + t * u[1]));
    const mk = (run, capEnd, zf) => { let [ra, rb] = run; if (capEnd) rb -= ROX_A3_EPS; if (rb <= ra + 1e-6) return null; const row = []; for (let j = 0; j <= 8; j++) { const R = ra + (rb - ra) * j / 8; row.push(V(R, deg, zf(R))); } return row; }; // roxlit: construction
    const first = runs.length ? runs[0] : null; // the first covered stretch (only 'void' — the model's own cut — may precede it; 'cloth' ends the ray)
    if (first && first[1] >= far - 1 / 2 - 1e-9) first[1] = far; // roxlit: construction — a run whose last sample reaches the horseshoe ends ON it
    let strip = null, shelf = null;
    if (first) { // split the first run by kind: the flange sheet over the model's rubber / rail region, a bed-level cloth patch over the shared hole's shelf part
      const kinds = []; for (let t = first[0]; t <= first[1] + 1e-9; t += 1 / 2) kinds.push([t, kindAt(Pm[0] + t * u[0], Pm[1] + t * u[1])]); // roxlit: construction
      const seg = (kind) => { const ix = kinds.map((kv, i) => [kv[1], i]).filter(([kk]) => kk === kind).map(([, i]) => i); if (ix.length < 2) return null; const i0 = ix[0], i1 = ix[ix.length - 1];
        return [i0 === 0 ? kinds[0][0] : (kinds[i0 - 1][0] + kinds[i0][0]) / 2, i1 === kinds.length - 1 ? kinds[i1][0] : (kinds[i1][0] + kinds[i1 + 1][0]) / 2]; }; // the two sheets meet at the midpoint between the last sample of one kind and the first of the other
      const sh = seg('shelf'), st = seg('strip'); const capEnd = first[1] < far - 1e-6; // roxlit: construction
      if (st) strip = mk(st, capEnd && (!sh || st[1] > sh[1]), (R) => zFl(deg, R));
      if (sh) shelf = mk(sh, capEnd && (!st || sh[1] > st[1]), () => 0);
    }
    sheetRows[0].push(strip); sheetRows[2].push(shelf);
    const second = h !== null && runs.length > 1 ? runs[1] : null; // the ear: the sheet resumes behind the cushion end (its exit onto the back line), out to the horseshoe
    sheetRows[1].push(second ? mk(second, second[1] < far - 1e-6, (R) => zFl(deg, R)) : null); // roxlit: construction
  }
  const emitRuns = (list, name, mat) => { const runs = []; let cur = []; for (const r of list) { if (r) cur.push(r); else if (cur.length) { runs.push(cur); cur = []; } } if (cur.length) runs.push(cur);
    runs.forEach((run, k) => { if (run.length > 1) group.add(dm(new THREE.Mesh(roxGrid(run, 'up'), mat), k === 0 ? name : `${name}-${k}`)); }); };
  emitRuns(sheetRows[0], `pocket-rox-corner-${tag}-flange`, mats.cup);
  emitRuns(sheetRows[1], `pocket-rox-corner-${tag}-flange-ear-0`, mats.cup);
  emitRuns(sheetRows[2], `pocket-rox-corner-${tag}-shelf`, mats.cloth); // the model's slab shelf (cloth) where our shared bed hole would show the pan beside the jaw (RC-22 / issue #643 item 3)
  // (3) casting: outer boundary polyline (long joint → long lip run → CP-15b superellipse → short lip run → short joint)
  const aJ = Ba - ROX_CORNER_SEAM.long, bJ = Bb - ROX_CORNER_SEAM.short; // CP-15 joints measured from the sharp corner
  const A = ROX_LIP_CORNER.a, nE = ROX_LIP_CORNER.n, ac = Ba - A, bc = Bb - A;
  const outer = [[aJ, ROX_CUSHION_BACK], [aJ, Bb], [ac, Bb]];
  for (let k = 1; k < 48; k++) { const th = Math.PI / 2 * (1 - k / 48); outer.push([ac + A * Math.pow(Math.cos(th), 2 / nE), bc + A * Math.pow(Math.sin(th), 2 / nE)]); } // roxlit: construction
  outer.push([Ba, bc], [Ba, bJ], [ROX_CUSHION_BACK, bJ]);
  const outerXZ = outer.map(([a, b]) => [xOf(a), zOf(b)]);
  const domeAt = (x, z) => { const zl = roxDomeZ('long', bOf(z)), zs = roxDomeZ('short', aOf(x)); const c = [zl, zs].filter((v) => v !== null); return c.length ? Math.min(...c) : null; };
  const angleTo = (a, b) => { const vx = xOf(a) - Pm[0], vz = zOf(b) - Pm[1]; return Math.atan2(vx * n[0] + vz * n[1], vx * d[0] + vz * d[1]) * 180 / Math.PI; };
  const phi0 = angleTo(ROX_CUSHION_BACK, bJ), phi1 = angleTo(aJ, ROX_CUSHION_BACK); // rays to the two joint / cushion-back corners
  const rowsC = [], wallTop = [];
  const steps = Math.max(8, Math.round((phi1 - phi0) * 2 / 3)); // one ray per 1.5 degrees
  for (let i = 0; i <= steps; i++) {
    const deg = phi0 + (phi1 - phi0) * i / steps; const u = dir(deg); const h = hs(deg);
    const rIn = Math.max(h ?? 0, exitT(u)); const [rOut, seg] = roxRayPolyline(Pm, u, outerXZ);
    if (!isFinite(rOut) || rOut <= rIn + 0.5) { rowsC.push(null); wallTop.push(null); continue; }
    const row = [];
    const proud = ROX_CASTING_PROUD * Math.min(1, Math.abs(deg) / Math.max(Math.abs(phi0), Math.abs(phi1))); // CP-15c: 0.357 proud AT THE JOINT plane; on the diagonal the casting crest (43.11) sits on the dome
    const rCrest = ROX_CASTING_CREST_R; // CP-15c crest u 143.92 on the diagonal − CP-12 cup u 45.32
    for (let k = 0; k <= 8; k++) { const R = rIn + (rOut - rIn) * k / 8; const x = Pm[0] + R * u[0], z = Pm[1] + R * u[1]; const zt = (domeAt(x, z) ?? ROX_CAP_TOP_Z_INNER) + proud; // roxlit: construction
      const t = Math.min(1, Math.max(0, (R - rIn) / Math.max(1, rCrest - rIn))); const zTop = ROX_CASTING_INNER_Z + (zt - ROX_CASTING_INNER_Z) * t; // CP-15: the top meets the horseshoe at the inner end face (41.4) and reaches the dome by the crest
      row.push(new THREE.Vector3(x / 1000, Math.min(zt, zTop) / 1000, z / 1000)); }
    rowsC.push(row);
    if (seg === 0 || seg === outerXZ.length - 2) { // joint planes: the wall starts / ends at the joint's lip corner, flush with the straight strip's end
      const [cxj, czj] = seg === 0 ? [xOf(aJ), zOf(Bb)] : [xOf(Ba), zOf(bJ)]; const zt = domeAt(cxj, czj);
      wallTop.push(new THREE.Vector3(cxj / 1000, ((zt ?? ROX_CAP_TOP_Z_INNER) + ROX_CASTING_PROUD) / 1000, czj / 1000));
    } else wallTop.push(row[8]);
  }
  const runs = []; let cur = [];
  for (const r of rowsC) { if (r) cur.push(r); else if (cur.length) { runs.push(cur); cur = []; } } if (cur.length) runs.push(cur);
  runs.forEach((run, k) => { if (run.length > 1) group.add(dm(new THREE.Mesh(roxGrid(run, 'up'), mats.metal), `pocket-rox-corner-${tag}-casting-top-${k}`)); });
  const wallRuns = []; cur = []; let last = null;
  for (const v of wallTop) { if (v) { if (!last || last.distanceTo(v) > 1e-6) cur.push([v, new THREE.Vector3(v.x, ROX_LIP_BOTTOM / 1000, v.z)]); last = v; } else if (cur.length) { wallRuns.push(cur); cur = []; last = null; } } if (cur.length) wallRuns.push(cur); // roxlit: construction
  wallRuns.forEach((run, k) => { if (run.length > 1) group.add(dm(new THREE.Mesh(roxGrid(run, 'out'), mats.metal), `pocket-rox-corner-${tag}-casting-lip-${k}`)); });
}
/** One side pocket assembly: half-ring liner + closed lower bucket, flange (with the SP-15 end tabs), aluminium rail piece between
 *  the SP-15 seams. p = kernel side pocket (scene metres). A3-PLAN: the SEEN liner hangs on the MODEL's cup centre (SP-12 a3,
 *  ROX_SIDE_CUP_CENTRE behind the nose line); the kernel axis keeps physics + the bed hole. The half-ring ends where its outer
 *  circle meets the cushion END FACE (vertical after roxCushionGeom) if that exists; with the MFR side mouth kept (SP-01, 127) it
 *  does not (the kernel back corner is 2.6 mm outside the circle), so the ring ends on the cushion BACK LINE — where the model's
 *  own ends sit — and the flange closes the residual strip (RC-29). Flange / tab / rail piece are rays from the cup centre clipped
 *  against the real cushion footprint and the bed hole (roxRayRuns): nothing starts at the kernel axis any more. */
function roxSidePocket(group, dm, G, p, mats, tag) {
  const W = G.W * 1000, sz = Math.sign(p.z); const Bb = (G.CW + G.render.frameWidth) * 1000;
  const bOf = (z) => (z - sz * W / 2) * sz; const Pm = [0, sz * (W / 2 + ROX_SIDE_CUP_CENTRE)]; // RC-32: the model's cup centre
  const Rin = ROX_CUP_R_IN.side, Rout = Rin + ROX_CUP_WALL; // SP-12 liner radius
  const dir = (deg) => { const r = deg * Math.PI / 180; return [Math.cos(r), sz * Math.sin(r)]; }; // theta 0 = +x, 90 = straight back into the rail
  const V = (r, deg, z) => { const u = dir(deg); return new THREE.Vector3((Pm[0] + r * u[0]) / 1000, z / 1000, (Pm[1] + r * u[1]) / 1000); };
  const mir = (deg) => Math.min(deg, 180 - deg);
  const rim = (deg) => (mir(deg) < ROX_SIDE_RIM[0][0] ? ROX_SIDE_RIM_END : roxLerp(ROX_SIDE_RIM, mir(deg)));
  // cushion ends in this pocket's frame (along = |x| from the pocket centre, behind = beyond the nose line) and the shared bed hole (RC-22)
  const E = roxCushionEnd(G, 'side'); const vc = bOf(Pm[1]);
  const inCushion = (x, z) => roxInsideEnd(E, Math.abs(x), bOf(z));
  const hole = { x: p.x * 1000, z: p.z * 1000, R: p.r * 1000 + ROX_BED_HOLE_REVEAL };
  const inHole = (x, z) => Math.hypot(x - hole.x, z - hole.z) < hole.R;
  const inModel = (x, z) => roxInsideModelEnd(E, ROX_SIDE_MODEL_FACE, Math.abs(x), bOf(z));
  // top where: behind the back line (rail region), inside the model's rubber but outside ours (the residual strip), or over the bed
  // hole behind the nose line; cloth (the shelf) in front of the nose line or beside the cup where the model itself shows cloth
  const inNotch = (x, z) => { const ax = Math.abs(x), v = bOf(z); const Nn = ROX_SIDE_NOTCH; // inside the model's slab notch (SP-14: the |x| < 49.81 arc): in front of its edge the model's slate shelf runs over the cup's own front; beside the cup the notch's steeper leg runs under the model's own rubber, where the between-ends cover applies
    return ax <= Nn.departX && Math.hypot(ax, v - Nn.arcCentreBack) < Nn.arcR; };
  const inCup = (x, z) => Math.hypot(x - Pm[0], z - Pm[1]) < Rout; // the cup's own opening
  // kinds: our rubber; inside the cup opening: 'void' where the model's notch is open, the model's slate shelf ('shelf', cloth patch at
  // bed level) where our shared hole reaches past the notch edge; outside the cup: 'strip' = the model's rubber or the rail region
  // (flange sheet), 'shelf' = over the shared hole where the model's shelf stands, 'cloth' = the shared bed sheet is the right top
  // the model-rubber-minus-ours wedge runs along the whole jaw face; only its part behind the model's back-top corner (SP-10 plane point,
  // 45.61 behind the nose line) is the between-ends top (flange); further forward the wedge is shelf beside a shorter rubber — cloth
  const inStrip = (x, z) => inModel(x, z) && bOf(z) >= ROX_SIDE_MODEL_FACE[1];
  const kindAt = (x, z) => inCushion(x, z) ? 'cushion' : inCup(x, z) ? (inNotch(x, z) ? 'void' : inHole(x, z) ? 'shelf' : 'cloth') : (bOf(z) >= E.CW || inStrip(x, z)) ? 'strip' : inHole(x, z) ? 'shelf' : 'cloth';
  const classify = (x, z) => { const k = kindAt(x, z); return k === 'strip' || k === 'shelf' ? 'top' : k; };
  // (0) where the ring ends: outer circle ∩ facing segment J→B if it exists, else the cushion back line (behind = CU-04)
  let thEnd = null; {
    const ex = E.B[0] - E.J[0], ey = E.B[1] - E.J[1]; const fx = E.J[0], fy = E.J[1] - vc;
    const qa = ex * ex + ey * ey, qb = 2 * (fx * ex + fy * ey), qc = fx * fx + fy * fy - Rout * Rout; const disc = qb * qb - 4 * qa * qc; // roxlit: construction
    if (disc >= 0) for (const sg of [-1, 1]) { const w = (-qb + sg * Math.sqrt(disc)) / (2 * qa); if (w >= 0 && w <= 1) thEnd = Math.atan2(E.J[1] + w * ey - vc, E.J[0] + w * ex) * 180 / Math.PI; }
  }
  // no contact under the MFR mouth (RC-29): the ring ends where the model's own ends sit (SP-12 end_y → 47.68 behind the nose), 1.9° below
  // the +x axis; the outer circle ∩ back line would be −1.44° — 0.4 mm apart, the model's is the read
  const th0 = thEnd !== null ? thEnd : Math.asin(Math.max(-1, Math.min(1, (ROX_SIDE_RING_END_V - vc) / Rout))) * 180 / Math.PI;
  const rows = [];
  const nR = Math.ceil((180 - 2 * th0) / 3); // roxlit: construction
  for (let k = 0; k <= nR; k++) { const deg = th0 + (180 - 2 * th0) * k / nR; rows.push([V(Rin, deg, ROX_SIDE_BUCKET.closedZ), V(Rin, deg, rim(deg)), V(Rout, deg, rim(deg)), V(Rout, deg, ROX_SIDE_BUCKET.closedZ)]); } // the ring above the closed oval; the bucket owns the floor → closedZ
  // closed lower bucket (SP-12): the back ring continues; the front is the R 51.5 half-circle about the model's own front centre (46.27 beyond the nose)
  const fc = [0, sz * (W / 2 + ROX_SIDE_BUCKET.frontCentreBack)];
  const rowsB = [];
  for (let deg = th0; deg <= 180 - th0 + 1e-6; deg += 3) { const u = dir(deg); const o = (r) => new THREE.Vector3((Pm[0] + r * u[0]) / 1000, 0, (Pm[1] + r * u[1]) / 1000); const pi = o(Rin), po = o(Rout); rowsB.push([new THREE.Vector3(pi.x, ROX_CUP_FLOOR_Z / 1000, pi.z), new THREE.Vector3(pi.x, (ROX_SIDE_BUCKET.closedZ - 1 / 10) / 1000, pi.z), new THREE.Vector3(po.x, (ROX_SIDE_BUCKET.closedZ - 1 / 10) / 1000, po.z), new THREE.Vector3(po.x, ROX_CUP_FLOOR_Z / 1000, po.z)]); } // back half of the closed oval: the ring's own circle continued down to the floor // roxlit: construction
  for (let deg = 180 - th0; deg <= 360 + th0 + 1e-6; deg += 3) { // roxlit: construction
    const u = dir(deg); const isFront = deg > 180 && deg < 360; const C = isFront ? fc : Pm; const R = isFront ? ROX_SIDE_BUCKET.frontR : Rout;
    const o = (r) => new THREE.Vector3((C[0] + r * u[0]) / 1000, 0, (C[1] + r * u[1]) / 1000);
    const pi = o(R - ROX_CUP_WALL), po = o(R);
    rowsB.push([new THREE.Vector3(pi.x, ROX_CUP_FLOOR_Z / 1000, pi.z), new THREE.Vector3(pi.x, (ROX_SIDE_BUCKET.closedZ - 1 / 10) / 1000, pi.z), new THREE.Vector3(po.x, (ROX_SIDE_BUCKET.closedZ - 1 / 10) / 1000, po.z), new THREE.Vector3(po.x, ROX_CUP_FLOOR_Z / 1000, po.z)]); // a tenth of a millimetre under the ring's bottom edge
  }
  const rowsF = [];
  for (let deg = 0; deg <= 360 + 1e-6; deg += 6) { const isFront = deg > 180 && deg < 360; const C = isFront ? fc : Pm; const R = (isFront ? ROX_SIDE_BUCKET.frontR : Rout) - ROX_CUP_WALL; const u = dir(deg); rowsF.push([V(ROX_CUP_DRAIN_R.side, deg, ROX_CUP_FLOOR_Z), new THREE.Vector3((C[0] + R * u[0]) / 1000, ROX_CUP_FLOOR_Z / 1000, (C[1] + R * u[1]) / 1000)]); } // roxlit: construction
  rowsB.push(rowsB[0]); rowsF.push(rowsF[0]); // close both fans: the front half (about fc) and the back half (about Pm) meet at the ring end, and an open seam there showed the pan from above
  group.add(dm(new THREE.Mesh(roxGrid(rows), mats.well), `pocket-rox-side-${tag}-cup-ring`));
  group.add(dm(new THREE.Mesh(roxGrid([[rows[0][0], rows[0][3]], [rows[0][1], rows[0][2]]]), mats.well), `pocket-rox-side-${tag}-cup-ring-end-0`)); // the wall's end faces (SP-12 half_ring end caps)
  group.add(dm(new THREE.Mesh(roxGrid([[rows[rows.length - 1][0], rows[rows.length - 1][3]], [rows[rows.length - 1][1], rows[rows.length - 1][2]]]), mats.well), `pocket-rox-side-${tag}-cup-ring-end-1`));
  group.add(dm(new THREE.Mesh(roxGrid(rowsB), mats.well), `pocket-rox-side-${tag}-cup-bucket`));
  group.add(dm(new THREE.Mesh(roxGrid(rowsF), mats.well), `pocket-rox-side-${tag}-cup-floor`));
  // flange (SP-15): flat top by theta, back slope from the rim over the model's u 51 → 62 band re-based onto the liner radius; below
  // theta 10 the tables clamp to their first row = the SP-15 end tab (x 50.2..61.1, flat 42.38); rays run from the ring's outer face
  // to the flange edge, clipped by the cushion footprint and the bed hole — past the ring end only while the ray still meets the
  // cushion end face (the between-ends strip), never over the open throat
  const flangeR = (deg) => roxLerp(ROX_SIDE_FLANGE_R, mir(deg)); const flat = (deg) => roxLerp(ROX_SIDE_FLANGE_FLAT, mir(deg));
  const zFl = (deg, R) => { const um = R - Rin + ROX_SIDE_INNER_R_MODEL; const g = Math.max(0, Math.min(1, (ROX_SIDE_FLANGE_SLOPE.flatU - um) / (ROX_SIDE_FLANGE_SLOPE.flatU - ROX_SIDE_FLANGE_SLOPE.rimU))); return flat(deg) + (rim(deg) - flat(deg)) * g; };
  const rowsG = [], rowsS = []; // flange rows / bed-level shelf rows per ray
  const angles = []; for (let deg = th0 - 90; deg <= 270 - th0 + 1e-6; deg += 1 / 2) angles.push(deg); // roxlit: construction (half-degree rays from a quarter turn before the ring end to a quarter turn after; the classifier decides where a sheet exists)
  for (const e of [th0, 180 - th0]) for (let dd = 0 - 4; dd <= 4 + 1e-9; dd += 1 / 10) angles.push(e + dd); // roxlit: construction (tenth-degree rays where the sheets meet the cushion back line)
  angles.sort((p, q) => p - q);
  for (const deg of angles) {
    const u = dir(deg); const beyond = deg < th0 || deg > 180 - th0;
    const R0 = beyond ? ROX_CUP_DRAIN_R.side : Rout; // past the ring end the ray starts at the drain: the model's slate shelf runs over the cup's own front, and there is no wall in the way
    const Ro = Math.max(Rout, flangeR(deg)); const runs = roxRayRuns(R0, Ro, (t) => classify(Pm[0] + t * u[0], Pm[1] + t * u[1]));
    if (!runs.length) { rowsG.push(null); rowsS.push(null); continue; } // the first covered stretch (only 'void' — the model's own notch — may precede it; 'cloth' ends the ray)
    const first = runs[0]; if (first[1] >= Ro - 1 / 2 - 1e-9) first[1] = Ro; const capEnd = first[1] < Ro - 1e-6; // roxlit: construction — a run whose last sample reaches the flange edge ends ON it
    const kinds = []; for (let t = first[0]; t <= first[1] + 1e-9; t += 1 / 2) kinds.push([t, kindAt(Pm[0] + t * u[0], Pm[1] + t * u[1])]); // roxlit: construction
    const seg = (kind) => { const ix = kinds.map((kv, i) => [kv[1], i]).filter(([kk]) => kk === kind).map(([, i]) => i); if (ix.length < 2) return null; const i0 = ix[0], i1 = ix[ix.length - 1];
      return [i0 === 0 ? kinds[0][0] : (kinds[i0 - 1][0] + kinds[i0][0]) / 2, i1 === kinds.length - 1 ? kinds[i1][0] : (kinds[i1][0] + kinds[i1 + 1][0]) / 2]; }; // the two sheets meet at the midpoint between the last sample of one kind and the first of the other
    const mk = (run, cap, zf) => { let [ra, rb] = run; if (cap) rb -= ROX_A3_EPS; if (rb <= ra + 1e-6) return null; const row = []; for (let k = 0; k <= 8; k++) { const R = ra + (rb - ra) * k / 8; row.push(V(R, deg, zf(R))); } return row; }; // roxlit: construction
    const sh = seg('shelf'), st = seg('strip');
    rowsG.push(st ? mk(st, capEnd && (!sh || st[1] > sh[1]), (R) => zFl(deg, R)) : null);
    rowsS.push(sh ? mk(sh, capEnd && (!st || sh[1] > st[1]), () => 0) : null);
  }
  { const emit = (list, name, mat) => { const runsG = []; let cur = []; for (const r of list) { if (r) cur.push(r); else if (cur.length) { runsG.push(cur); cur = []; } } if (cur.length) runsG.push(cur);
      runsG.forEach((run, k) => { if (run.length > 1) group.add(dm(new THREE.Mesh(roxGrid(run, 'up'), mat), k === 0 ? name : `${name}-${k}`)); }); };
    emit(rowsG, `pocket-rox-side-${tag}-flange`, mats.cup);
    emit(rowsS, `pocket-rox-side-${tag}-shelf`, mats.cloth); } // the model's slab shelf (cloth) where our shared bed hole would show the pan beside the jaw (RC-22 / issue #643 item 3)
  // rail piece between the seams: polar heightfield from the flange edge out to the lip line or the seam planes, never over the
  // cushion (the sweep now starts at the ring end, so the piece begins at the cushion BACK LINE, as the model's does)
  const half = ROX_SIDE_SEAM_HALF; const outer = [[-half, ROX_CUSHION_BACK], [-half, Bb], [half, Bb], [half, ROX_CUSHION_BACK], [-half, ROX_CUSHION_BACK]].map(([x, b]) => [x, sz * (W / 2 + b)]);
  const rowsC = [], wallTop = []; const phi0 = th0, phi1 = 180 - th0; const steps = Math.max(8, Math.round((phi1 - phi0) * 2 / 3)); // one ray per 1.5 degrees
  const pieceAngles = []; for (let i = 0; i <= steps; i++) pieceAngles.push(phi0 + (phi1 - phi0) * i / steps);
  for (let dd = 1 / 10; dd < 4; dd += 1 / 10) { pieceAngles.push(phi0 + dd, phi1 - dd); } pieceAngles.sort((p, q) => p - q); // tenth-degree rays where the piece meets the cushion back line (a coarse fan leaves a bed-cloth wedge there) // roxlit: construction
  for (const deg of pieceAngles) {
    const u = dir(deg);
    const rIn = Math.max(Rout, flangeR(deg)); const [rOut, seg] = roxRayPolyline(Pm, u, outer);
    if (!isFinite(rOut) || rOut <= rIn + 0.5) { rowsC.push(null); wallTop.push(null); continue; }
    const runs = roxRayRuns(rIn, rOut, (t) => classify(Pm[0] + t * u[0], Pm[1] + t * u[1]));
    if (!runs.length || runs[0][0] > rIn + 1) { rowsC.push(null); wallTop.push(null); continue; }
    let [ra, rb] = runs[0]; if (rb >= rOut - 1 / 2 - 1e-9) rb = rOut; if (rb <= ra + 0.5) { rowsC.push(null); wallTop.push(null); continue; } // roxlit: construction — a run whose last half-millimetre sample reaches the outer polygon ends ON it (the lip wall stands there)
    const row = [];
    for (let k = 0; k <= 8; k++) { const R = ra + (rb - ra) * k / 8; const x = Pm[0] + R * u[0], z = Pm[1] + R * u[1]; const zt = roxDomeZ('long', bOf(z)); row.push(new THREE.Vector3(x / 1000, ((zt ?? ROX_CAP_TOP_Z_INNER) + ROX_SIDE_PIECE_PROUD) / 1000, z / 1000)); } // roxlit: construction
    rowsC.push(row);
    wallTop.push(seg === 1 && rb >= rOut - 1e-6 ? row[8] : null); // roxlit: construction — only the lip-line segment carries a visible outer wall; the seam planes are hidden inside the cap ends
  }
  const runs = []; let cur = [];
  for (const r of rowsC) { if (r) cur.push(r); else if (cur.length) { runs.push(cur); cur = []; } } if (cur.length) runs.push(cur);
  runs.forEach((run, k) => { if (run.length > 1) group.add(dm(new THREE.Mesh(roxGrid(run, 'up'), mats.metal), `pocket-rox-side-${tag}-piece-top-${k}`)); });
  const wallRuns = []; cur = [];
  for (const v of wallTop) { if (v) cur.push([v, new THREE.Vector3(v.x, ROX_LIP_BOTTOM / 1000, v.z)]); else if (cur.length) { wallRuns.push(cur); cur = []; } } if (cur.length) wallRuns.push(cur);
  wallRuns.forEach((run, k) => { if (run.length > 1) group.add(dm(new THREE.Mesh(roxGrid(run, 'out'), mats.metal), `pocket-rox-side-${tag}-piece-lip-${k}`)); });
}
